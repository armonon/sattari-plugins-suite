cmake_minimum_required(VERSION 3.22)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "Self-contained Windows runtime")
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/SattariDeploymentTarget.cmake)
project(SattariReleaseCandidates VERSION 0.2.1 LANGUAGES C CXX)

enable_testing()
get_filename_component(SATTARI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SATTARI_COPY_PLUGINS_AFTER_BUILD OFF CACHE BOOL "Install plugin bundles after building")
set(SATTARI_USE_ONNX OFF CACHE BOOL "Optional neural tools are outside this candidate build")
set(SATTARI_USE_RUBBERBAND ON CACHE BOOL "Preserve Vocal's production pitch engine")
if(NOT SATTARI_USE_RUBBERBAND OR NOT EXISTS "${SATTARI_ROOT}/third_party/rubberband/single/RubberBandSingle.cpp")
    message(FATAL_ERROR "The candidate build requires the vendored Rubber Band engine")
endif()

file(READ "${CMAKE_CURRENT_LIST_DIR}/products.json" products)
string(JSON count LENGTH "${products}")
math(EXPR last "${count} - 1")
set(candidate_targets)
set(contracts SattariArpTimingContract SubConjurerContract SattariMixProductContract
    SattariMixAutoMixContract SattariVocalProductContract SattariCreateProductContract)
foreach(index RANGE ${last})
    foreach(field slug target version)
        string(JSON ${field} GET "${products}" ${index} ${field})
    endforeach()
    add_subdirectory("${SATTARI_ROOT}/plugins/${slug}" "${slug}")
    get_target_property(actual_version ${target} JUCE_VERSION)
    if(NOT actual_version STREQUAL version)
        message(FATAL_ERROR "${slug}: manifest ${version} differs from plugin ${actual_version}")
    endif()
    foreach(format AU VST3 Standalone)
        if(TARGET ${target}_${format})
            list(APPEND candidate_targets ${target}_${format})
        endif()
    endforeach()
    list(APPEND contracts ${target}UIContract)
    add_test(NAME ${target}UIContract COMMAND ${target}UIContract "${CMAKE_BINARY_DIR}/validation/screenshots")
endforeach()

add_executable(SattariMixRackRegression EXCLUDE_FROM_ALL "${SATTARI_ROOT}/core/tests/rack_host_contract.cpp")
target_include_directories(SattariMixRackRegression PRIVATE $<TARGET_PROPERTY:SattariMix,INCLUDE_DIRECTORIES>)
target_compile_definitions(SattariMixRackRegression PRIVATE $<TARGET_PROPERTY:SattariMix,COMPILE_DEFINITIONS>)
target_compile_features(SattariMixRackRegression PRIVATE cxx_std_17)
target_link_libraries(SattariMixRackRegression PRIVATE SattariMix)
list(APPEND contracts SattariMixRackRegression)
foreach(contract IN LISTS contracts)
    if(NOT contract MATCHES "UIContract$")
        add_test(NAME ${contract} COMMAND ${contract})
    endif()
endforeach()
add_custom_target(SattariCandidateBuild DEPENDS ${candidate_targets} ${contracts})
